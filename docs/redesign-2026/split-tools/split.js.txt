/* One-shot splitter: app.js -> app.js (core) + ui/*.js.
 * usage: node split.js <repoRoot> <origAppJs> [--write]
 * Reads the ORIGINAL app.js from <origAppJs> (so it is re-runnable), writes into <repoRoot>.
 */
const fs = require("fs");
const path = require("path");
const acorn = require("acorn");

const root = process.argv[2];
const origPath = process.argv[3];
const WRITE = process.argv.includes("--write");
const src = fs.readFileSync(origPath, "utf8");
const ast = acorn.parse(src, { ecmaVersion: "latest", sourceType: "script", locations: true });

// start line of a top-level node -> file. Last entry with start <= line wins.
const RANGES = [

const NAME_OVERRIDES = { announce: "core" };
function fileFor(line) {
  let f = "core";
  for (const [s, name] of RANGES) if (s <= line) f = name;
  return f;
}

// build segments: [start incl. leading trivia, end]
const segs = [];
let prevEnd = 0;
for (const n of ast.body) {
  segs.push({ node: n, from: prevEnd, to: n.end, line: n.loc.start.line });
  prevEnd = n.end;
}
const tailTrivia = src.slice(prevEnd);

const out = {};
for (const s of segs) {
  s.file = (s.node.type === "FunctionDeclaration" && NAME_OVERRIDES[s.node.id.name]) || fileFor(s.line);
  (out[s.file] = out[s.file] || []).push(s);
}

const nameOf = (n) =>
  n.type === "FunctionDeclaration" ? n.id.name :
  n.type === "VariableDeclaration" ? n.declarations.map((d) => d.id.name || "{}").join(",") : "##" + n.type;

const report = {};
for (const [file, list] of Object.entries(out)) {
  report[file] = { nodes: list.length, lines: list.reduce((a, s) => a + (s.node.loc.end.line - s.node.loc.start.line + 1), 0) };
}
console.log(JSON.stringify(report, null, 1));

// emit
const files = {};
for (const [file, list] of Object.entries(out)) {
  let body = list.map((s) => src.slice(s.from, s.to)).join("");
  if (file === "core") { body += tailTrivia; files[file] = body; continue; }
  const header =
    `/* ${file} — ${TITLES[file] || file}\n` +
    `   A CLASSIC script like app.js, not a module: it shares app.js's globals and\n` +
    `   is loaded by index.html after app.js, in the order listed in\n` +
    `   docs/redesign-2026/split-notes.md. Nothing here may call into the app at load, so\n` +
    `   ui/boot.js is what starts init(). */\n`;
  // leading trivia of the first segment began right after the previous (different-file) node; keep it
  files[file] = header + body.replace(/^\n+/, "\n") + "\n";
}
// moved-name manifest (for later checks)
const manifest = {};
for (const [file, list] of Object.entries(out)) manifest[file] = list.map((s) => nameOf(s.node));
fs.writeFileSync(path.join(path.dirname(origPath), "split-manifest.json"), JSON.stringify(manifest, null, 1));
if (WRITE) {
  fs.mkdirSync(path.join(root, "ui"), { recursive: true });
  for (const [file, body] of Object.entries(files)) {
    fs.writeFileSync(path.join(root, file === "core" ? "app.js" : file), body);
  }
  console.log("written", Object.keys(files).length, "files");
}
